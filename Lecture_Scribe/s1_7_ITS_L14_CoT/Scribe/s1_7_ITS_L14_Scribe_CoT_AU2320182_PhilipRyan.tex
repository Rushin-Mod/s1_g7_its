\section*{Q1. Random Point on a Line Segment}

\textbf{Problem Interpretation}

A point is chosen uniformly at random on a line segment of length $L$.

Let:
\[
x \in (0, L)
\]

Two segments formed: $x$ and $L - x$

Define the ratio:
\[
R = \frac{\max(x, L - x)}{\min(x, L - x)}
\]

We need:
\[
P\left(R < \frac{1}{4}\right)
\]

\textbf{Step-by-Step Reasoning}

Case 1: $x \le \frac{L}{2}$

Shorter segment = $x$ \\
Longer segment = $L - x$

\[
\frac{x}{L - x} < \frac{1}{4}
\]

\[
4x < L - x
\Rightarrow 5x < L
\Rightarrow x < \frac{L}{5}
\]

Case 2: $x > \frac{L}{2}$

Shorter segment = $L - x$ \\
Longer segment = $x$

\[
\frac{L - x}{x} < \frac{1}{4}
\]

\[
4L - 4x < x
\Rightarrow 5x > 4L
\Rightarrow x > \frac{4L}{5}
\]

\textbf{Valid Region}

\[
(0, \frac{L}{5}) \cup (\frac{4L}{5}, L)
\]

Total valid length:
\[
\frac{L}{5} + \frac{L}{5} = \frac{2L}{5}
\]

\textbf{Final Probability}

\[
P = \frac{2L/5}{L} = \frac{2}{5}
\]

\section*{Q2. Bayesian Classification with Normal Distributions}

\textbf{Given}

\[
X|W \sim N(4,4), \quad X|B \sim N(6,9)
\]

\[
P(B) = \alpha, \quad P(W) = 1 - \alpha
\]

Observed:
\[
X = 5
\]

Condition:
\[
P(B|X=5) = P(W|X=5) = \frac{1}{2}
\]

\textbf{Applying Bayes’ Theorem}

\[
\alpha f(5|B) = (1 - \alpha) f(5|W)
\]

\textbf{Normal Density Values}

\[
f(5|B) = \frac{1}{3\sqrt{2\pi}} e^{-1/18}, \quad
f(5|W) = \frac{1}{2\sqrt{2\pi}} e^{-1/8}
\]

\textbf{Substitution}

\[
\frac{\alpha e^{-1/18}}{3\sqrt{2\pi}} =
\frac{(1 - \alpha)e^{-1/8}}{2\sqrt{2\pi}}
\]

Simplifies to:

\[
\frac{3\alpha}{3\alpha + 2(1-\alpha)e^{-5/72}} = \frac{1}{2}
\]

\textbf{Solving}

\[
6\alpha = 3\alpha + 2(1-\alpha)(0.93)
\]

\[
3\alpha = 1.86 - 1.86\alpha
\Rightarrow 4.86\alpha = 1.86
\Rightarrow \alpha \approx 0.3827
\]

\textbf{Final Answer}

\[
\alpha \approx 0.38
\]

\section*{Q3. Optimal Location Problem}

\subsection*{(a) Uniform Distribution}

\[
X \sim \text{Uniform}(0,A), \quad f_X(x) = \frac{1}{A}
\]

\[
E[|X - a|] = \int_0^A |x - a| \frac{1}{A} dx
\]

Breaking absolute value:

\[
E[|X - a|] =
\int_0^a (a - x)\frac{1}{A} dx +
\int_a^A (x - a)\frac{1}{A} dx
\]

After simplification:

\[
\frac{d}{da} \left(\frac{2a^2 - 2aA + A^2}{2A}\right)
= \frac{4a - 2A}{2A}
\]

Minimization:

\[
4a - 2A = 0 \Rightarrow a = \frac{A}{2}
\]

\textbf{Conclusion:} midpoint.

\subsection*{(b) Exponential Distribution}

\[
X \sim \text{Exponential}(\lambda), \quad f_X(x) = \lambda e^{-\lambda x}
\]

\[
E[|X - a|] =
\int_0^a (a - x)\lambda e^{-\lambda x} dx +
\int_a^\infty (x - a)\lambda e^{-\lambda x} dx
\]

Derivative:

\[
\frac{d}{da} E[|X - a|] = 1 - 2e^{-\lambda a}
\]

Minimization:

\[
1 - 2e^{-\lambda a} = 0
\Rightarrow e^{-\lambda a} = \frac{1}{2}
\Rightarrow a = \frac{\ln 2}{\lambda}
\]

\section*{Q5. Conditional Distribution of Poisson Variables}

\[
X \sim \text{Poisson}(\lambda_1), \quad Y \sim \text{Poisson}(\lambda_2)
\]

\[
P(X = k | X + Y = n) =
\frac{P(X=k, Y=n-k)}{P(X+Y=n)}
\]

After simplification:

\[
\binom{n}{k}
\left(\frac{\lambda_1}{\lambda_1 + \lambda_2}\right)^k
\left(\frac{\lambda_2}{\lambda_1 + \lambda_2}\right)^{n-k}
\]

\section*{Q6. Transformation of Random Variable}

\[
X \sim \text{Uniform}[-2,2]
\]

\[
F_Y(y) = P(Y \le y)
\]

Piecewise:

\[
F_Y(y) =
\begin{cases}
0 & y < -2 \\
\frac{y+2}{4} & -2 \le y < -1 \\
\frac{1}{4} & -1 \le y < 0 \\
\frac{1}{2} & y = 0 \\
\frac{3}{4} & 0 < y < 1 \\
\frac{y+2}{4} & 1 \le y \le 2 \\
1 & y > 2
\end{cases}
\]

\[
f_Y(y) =
\begin{cases}
\frac{1}{4} & -2 < y < -1 \\
\frac{1}{4} & 1 < y < 2 \\
0 & \text{otherwise}
\end{cases}
\]

\section*{Q7. Stock Model Approximation}

\[
S_{1000} = S_0 u^k d^{1000-k}
\]

Condition:

\[
S_{1000} \ge 1.30 S_0
\]

Log transformation:

\[
k \ln u + (1000-k)\ln d \ge \ln(1.30)
\]

\[
k \ge \frac{\ln(1.30) - 1000\ln d}{\ln u - \ln d}
\approx 469.4 \Rightarrow k \ge 470
\]

\[
k \sim \text{Binomial}(1000, 0.52)
\]

\[
P(k \ge 470) =
\sum_{k=470}^{1000}
\binom{1000}{k}
(0.52)^k (0.48)^{1000-k}
\]

\section*{Q9. Distribution of $Z = X^2 + Y^2$}

\[
F_Z(z) = P(X^2 + Y^2 \le z)
\]

\[
F_Z(z) =
\int_{-z}^{z}
\int_{-\sqrt{z^2 - x^2}}^{\sqrt{z^2 - x^2}}
f_X(x)f_Y(y)\, dy\, dx
\]

\[
f_Z(z) = \frac{d}{dz}F_Z(z)
\]

Using Leibniz rule:

\[
f_Z(z) =
z \int_{-z}^{z}
f_X(x)\left[
f_Y(\sqrt{z^2 - x^2}) + f_Y(-\sqrt{z^2 - x^2})
\right] dx
\]

\section*{Q10. Joint Distribution of Maxima}

\[
M_n = \max(X_1,\dots,X_n)
\]

\[
P(M_n \le x) = F(x)^n
\]

\[
M_{n+1} = \max(M_n, X_{n+1})
\]

\[
P(M_n \le x, M_{n+1} \le y) =
\begin{cases}
F(x)^n F(y) & x \le y \\
F(y)^{n+1} & x > y
\end{cases}
\]
