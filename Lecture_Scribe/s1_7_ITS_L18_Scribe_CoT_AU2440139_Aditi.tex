\section{Marginal PMF}

\subsection{Definition}
Joint PMF:
\[
p_{XY}(x,y) = P(X=x, Y=y)
\]

Marginal PMFs:
\[
p_X(x) = \sum_y p_{XY}(x,y), \quad
p_Y(y) = \sum_x p_{XY}(x,y)
\]

\textbf{Key Idea:} Sum over the other variable.

\subsection{From Joint PMF Table}
\[
p_X(x_i) = \sum_j p_{ij}, \quad
p_Y(y_j) = \sum_i p_{ij}
\]

\[
\sum_i p_X(x_i) = 1, \quad
\sum_j p_Y(y_j) = 1
\]

\subsection{Worked Example}
\begin{center}
\begin{tabular}{c|cc}
\toprule
 & $y=0$ & $y=1$ \\
\midrule
$x=0$ & $1/8$ & $1/4$ \\
$x=1$ & $3/8$ & $1/4$ \\
\bottomrule
\end{tabular}
\end{center}

\textbf{Step 1:}
\[
p_X(0) = \frac{1}{8} + \frac{1}{4} = \frac{3}{8}
\]
\[
p_X(1) = \frac{3}{8} + \frac{1}{4} = \frac{5}{8}
\]

\textbf{Step 2:}
\[
p_Y(0) = \frac{1}{8} + \frac{3}{8} = \frac{1}{2}
\]
\[
p_Y(1) = \frac{1}{4} + \frac{1}{4} = \frac{1}{2}
\]

\section{Correlation}

\subsection{Definition}
\[
R_{XY} = E[XY]
\]

\textbf{Interpretation:} Average value of the product $XY$.

\textbf{Limitation:} Not centered about means.

% ====================================================
\section{Covariance}

\subsection{Definition}
\[
\text{Cov}(X,Y) = E[(X-\mu_X)(Y-\mu_Y)]
\]

\subsection{Derivation}
\[
\text{Cov}(X,Y) = E[(X-\mu_X)(Y-\mu_Y)]
\]
\[
= E[XY - X\mu_Y - Y\mu_X + \mu_X\mu_Y]
\]
\[
= E[XY] - \mu_Y E[X] - \mu_X E[Y] + \mu_X\mu_Y
\]
\[
= E[XY] - E[X]E[Y]
\]

\subsection{Properties}
\begin{itemize}[noitemsep]
\item $\text{Cov}(X,Y) = \text{Cov}(Y,X)$
\item $\text{Cov}(X,X) = \text{Var}(X)$
\item $\text{Cov}(aX+b, cY+d) = ac\,\text{Cov}(X,Y)$
\item Independence $\Rightarrow \text{Cov}(X,Y)=0$
\end{itemize}

\subsection{Example}
Given:
\[
Y = -2X + 3
\]

\[
XY = -2X^2 + 3X
\]

\[
E[Y] = -2E[X] + 3
\]

\[
\text{Cov}(X,Y) = E[-2X^2 + 3X] - E[X](-2E[X]+3)
\]

\[
= -2E[X^2] + 2(E[X])^2
\]

\[
= -2\text{Var}(X)
\]

\section{Pearson Correlation Coefficient}

\subsection{Definition}
\[
\rho_{XY} = \frac{\text{Cov}(X,Y)}{\sigma_X \sigma_Y}
\]

\[
-1 \le \rho_{XY} \le 1
\]

\section{Continuous Example}

Given:
\[
f_{XY}(x,y) =
\begin{cases}
\frac{xy}{96}, & 0<x<4,\ 1<y<5 \\
0, & \text{otherwise}
\end{cases}
\]

\subsection{Step 1:}
\[
E[X] = \frac{1}{8}\int_0^4 x^2 dx = \frac{8}{3}
\]

\subsection{Step 2:}
\[
E[Y] = \frac{1}{12}\int_1^5 y^2 dy = \frac{31}{9}
\]

\subsection{Step 3:}
\[
E[XY] = E[X]E[Y] = \frac{248}{27}
\]

\subsection{Step 4:}
\[
E[2X+3Y] = \frac{47}{3}
\]

\subsection{Step 5:}
\[
E[X^2] = 8
\]
\[
\text{Var}(X) = \frac{8}{9}
\]

\subsection{Step 6:}
\[
E[Y^2] = 13
\]
\[
\text{Var}(Y) = \frac{92}{81}
\]

\subsection{Step 7:}
\[
\text{Cov}(X,Y) = 0
\]

\subsection{Step 8:}
\[
\rho_{XY} = 0
\]

\section{Joint Gaussian Random Variables}

\subsection{Definition}
A pair $(X,Y)$ is jointly Gaussian if the joint density is bivariate Gaussian.

\subsection{PDF}
\[
f_{XY}(x,y) =
\frac{1}{2\pi\sigma_X\sigma_Y\sqrt{1-\rho^2}}
\exp\left(
\frac{-1}{2(1-\rho^2)}
\left[
\left(\frac{x-\mu_X}{\sigma_X}\right)^2 +
\left(\frac{y-\mu_Y}{\sigma_Y}\right)^2 -
2\rho \left(\frac{x-\mu_X}{\sigma_X}\right)\left(\frac{y-\mu_Y}{\sigma_Y}\right)
\right]
\right)
\]
